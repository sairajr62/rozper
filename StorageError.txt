Commands when storage space gets full

2. # Clear Next.js cache (biggest win)
Remove-Item -Recurse -Force "C:\Users\OTHER-USERS\Downloads\b_i8jCFfXRN2t\b_i8jCFfXRN2t\.next"
3.# Clear npm cache

js/** @type {import('next').NextConfig} */

  },
